package com.office.ex;

import jakarta.servlet.RequestDispatcher;
import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;

@WebServlet("*.vc")
public class VaccineController extends HttpServlet {
	private static final long serialVersionUID = 1L;
       
	private String CLASS_NAME = "[VaccineController] ";
	
    public VaccineController() {
        super();
    }


	protected void doGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
		
		String requestURI = request.getRequestURI();
		String contextPath = request.getContextPath();
		String command = requestURI.substring(contextPath.length());
		String nextPage = null;
		VaccineService vaccineService = null;
		
		switch (command) {
		case VaccineConfig.VACCINE_RESISTER_FORM:
			System.out.println(CLASS_NAME.concat(VaccineConfig.VACCINE_RESISTER_FORM));
			
			nextPage = generateView("vaccine_register_form");
			
			break;
			
		case VaccineConfig.VACCINE_RESISTER_CONFIRM:
			System.out.println(CLASS_NAME.concat(VaccineConfig.VACCINE_RESISTER_CONFIRM));
			
			vaccineService = new VaccineService();
			int resultForRegiste = vaccineService.registeVaccineInjection(request, response);
			
			if (resultForRegiste > 0) {
				System.out.println(CLASS_NAME.concat("VACCINE INJECTION SUCCESS!!"));
				nextPage = generateView("vaccine_register_ok");
				
			} else {
				System.out.println(CLASS_NAME.concat("VACCINE INJECTION FAIL!!"));
				nextPage = generateView("vaccine_register_ng");
			}
			
			break;
			
		case VaccineConfig.VACCINE_LIST:
			System.out.println(CLASS_NAME.concat(VaccineConfig.VACCINE_LIST));
			
			break;
		}
		
		RequestDispatcher dispatcher = request.getRequestDispatcher(nextPage);
		dispatcher.forward(request, response);
		
	}


	protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
		request.setCharacterEncoding("UTF-8");
		doGet(request, response);
	}
	
	private String generateView(String viewName) {

		return VaccineConfig.VIEW_PREFIX
				.concat(viewName)
				.concat(VaccineConfig.VIEW_SUFFIX);
		
	}
	
	
}

public class VaccineService {

	final private String CLASS_NAME = "[VaccineService] ";
	
	private VaccineDao vaccineDao = new VaccineDao();
	
	public int registeVaccineInjection(HttpServletRequest request, HttpServletResponse response) {
		System.out.println(CLASS_NAME.concat("registeVaccineInjection()"));
		
		String fvp_name = request.getParameter("fvp_name");
		String ss_number = request.getParameter("ss_number");
		String fvp_mail = request.getParameter("fvp_mail");
		String fvp_phone = request.getParameter("fvp_phone");
		String fvp_location_no = request.getParameter("fvp_location_no");
		String fv_type = request.getParameter("fv_type");
		
		VaccineDto vaccineDto = 
				new VaccineDto(fvp_name, ss_number, fvp_mail, fvp_phone, fvp_location_no, fv_type);
		
		int result = vaccineDao.insertVaccineInjection(vaccineDto);
		
		if (result > 0) {
			System.out.println(CLASS_NAME.concat("VACCINE INJECTION SUCCESS!!"));
			
		} else {
			System.out.println(CLASS_NAME.concat("VACCINE INJECTION FAIl!!"));
			
		}
		
		return result;
		
	}

}

package com.office.ex;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;

public class VaccineDao implements IVaccineDao {

	final private String CLASS_NAME = "[VaccineDao] ";
	
	@Override
	public int insertVaccineInjection(VaccineDto vaccineDto) {
		System.out.println(CLASS_NAME.concat("insertVaccineInjection()"));
		
		Connection conn = null;
		PreparedStatement pstmt = null;
		int result = -1;
		
		try {
			Class.forName(DRIVER);
			conn = DriverManager.getConnection(URL, USER, PASSWORD);
			String sql = "INSERT INTO tbl_flu_vaccin_person("
							+ "fvp_name, "
							+ "ss_number, "
							+ "fvp_mail, "
							+ "fvp_phone, "
							+ "fvp_location_no, "
							+ "fv_type) "
						+ "VALUES(?, ?, ?, ?, ?, ?)";
			pstmt = conn.prepareStatement(sql);
			pstmt.setString(1, vaccineDto.getFvp_name());
			pstmt.setString(2, vaccineDto.getSs_number());
			pstmt.setString(3, vaccineDto.getFvp_mail());
			pstmt.setString(4, vaccineDto.getFvp_phone());
			pstmt.setString(5, vaccineDto.getFvp_location_no());
			pstmt.setString(6, vaccineDto.getFv_type());
			result = pstmt.executeUpdate();
					
		} catch (Exception e) {
			e.printStackTrace();
			
		} finally {
			try {
				if (pstmt != null) pstmt.close();
				if (conn != null) conn.close();
				
			} catch (Exception e2) {
				e2.printStackTrace();
				
			}
			
		}
		
		return result;
		
	}

}
