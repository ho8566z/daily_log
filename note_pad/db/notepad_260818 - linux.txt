##MySQL 설치 절차

####1. MySQL 다운로드

	MySQL 공식 홈페이지 접속
	MySQL Installer for Windows 다운로드
	Community 버전 선택
	
	
####2. MySQL Installer 실행

	설치 유형 선택
	Developer Default 권장
	필요한 MySQL 구성 요소 자동 설치
	
	
####3. MySQL Server 설정

	Config Type 설정
	TCP/IP 포트 설정 → 기본값 3306
	Windows Service 설정
	
	
####4. Root 계정 설정

	root 비밀번호 설정
	비밀번호는 MySQL 관리자 계정 접속에 사용
	
	
####5. 설치 진행

	설정 내용 확인
	Execute 클릭
	설치 완료
	
	
####6. MySQL 접속 확인

	MySQL Workbench 실행
	Local instance MySQL 접속
	root 계정과 설정한 비밀번호 입력

CREATE TABLE TBL_MEMBER (
    mNo        INT          NOT NULL AUTO_INCREMENT UNIQUE COMMENT '자동 생성 회원번호',
    mId        VARCHAR(50)  NOT NULL UNIQUE COMMENT '회원 ID',
    mPw        VARCHAR(100) NOT NULL COMMENT '회원 비밀번호',
    mMail      VARCHAR(50)  NOT NULL COMMENT '회원 메일주소',
    mPhone     VARCHAR(50)           COMMENT '회원 연락처',
    m_reg_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '회원 등록일',
    m_mod_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP 
			   ON UPDATE CURRENT_TIMESTAMP COMMENT '회원 수정일',

    PRIMARY KEY (mNo)
);


-- TABLE 구조확인 방법
DESC TBL_MEMBER;


-- TABLE의 데이터 조회(READ)
SELECT * FROM TBL_MEMBER;


-- TABLE의 데이터 부분조회(READ)
SELECT mNo NO, mId ID, m_reg_date REG_DATE FROM TBL_MEMBER;


-- WHERE == IF문 -> 조건절
SELECT 
	* 
FROM 
	TBL_MEMBER 
WHERE 
	mId = 'chanho' && 
    mPw = '5678';

SELECT 
	COUNT(*) CNT
FROM 
	TBL_MEMBER 
WHERE 
	mNo >= '2';
