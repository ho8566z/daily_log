## 이벤트 위임(Event Delegation) 이란?

- 버튼/링크 하나하나에 이벤트를 등록하는 대신, 상위 요소인 document 하나에만 이벤트 리스너를 걸어두고 event.target으로 실제로 클릭된 요소가 무엇인지 판별하는 방식이다.
- 달력의 "write" 버튼이나 일정 제목처럼, 페이지가 새로고침될 때마다(달이 바뀔 때마다) DOM이 통째로 새로 만들어지는 요소에 특히 유리하다. 매번 새로 생긴 요소마다 이벤트를 다시 걸어줄 필요 없이, document에 딱 한 번만 등록해두면 이후 생기는 요소까지 전부 자동으로 처리된다.
- event.target.matches(selector)로 "지금 클릭된 요소가 내가 원하는 선택자와 일치하는가"를 검사한다.

#### [ planner.js - initEvents() ]

```jsx
document.addEventListener('click', function (event) {

    // 이전달 버튼 클릭 시
    if (event.target.matches('#section_wrap .btn_pre')) {
        setPreMonth();
    }

    // 달력의 일정 등록(write) 버튼 클릭 시

        showWritePlanView(current_year, current_month + 1, date);

    // 일정 제목 클릭 시 상세보기
    if (event.target.matches('#table_calender a.title')) {
        fetchGetPlan(event.target.getAttribute('data-no'));
    }

});
```

## Date 객체로 달력 그리드 만들기

- new Date(year, month, 0) 처럼 일(day)에 0을 넣으면, JS Date 객체가 자동으로 "month의 전달, 마지막 날"로 보정해준다. 즉 new Date(2026, 9, 0)은 9월(month 인덱스 9 = 10월)의 0번째 날 = 9월의 마지막 날이 된다. 이 성질을 이용해서 "이번 달의 마지막 날짜"를 구한다.
- getDay()로 이번 달 1일이 무슨 요일(0~6, 0=일요일)인지 구하고, 그 요일 수만큼 앞에 빈 칸(0)을 채운 뒤 날짜를 순서대로 채워서 6주(7일 × 6 = 42칸) 짜리 달력 격자를 만든다.

#### [ planner.js - addCalenderTr() ]

```jsx
let thisCalenderStart = new Date(current_year, current_month, 1);
let thisCalenderStartDay = thisCalenderStart.getDay();     // 이번 달 1일의 요일

let thisCalenderEnd = new Date(current_year, current_month + 1, 0);
let thisCalenderEndDate = thisCalenderEnd.getDate();       // 이번 달의 마지막 날짜

let dates = Array();
let dateCnt = 1;
for (let i = 0; i < 42; i++) {
    if (i < thisCalenderStartDay || dateCnt > thisCalenderEndDate) {
        dates[i] = 0;      // 빈 칸
    } else {
        dates[i] = dateCnt;
        dateCnt++;
    }
}
```

## async / await + fetch API로 서버 통신하기

- fetch()는 기본적으로 Promise를 반환하는데, async 함수 안에서 await을 붙이면 마치 동기 코드처럼 위에서 아래로 순서대로 읽을 수 있다.
- fetch는 서버가 4xx/5xx를 응답해도 네트워크 자체는 성공했기 때문에 reject 하지 않는다. 그래서 response.ok를 직접 검사해서 실패 시 throw new Error(...)로 강제로 에러를 던지고, catch 블록에서 잡아 처리하는 패턴이 자주 쓰인다.
- try(정상 처리) / catch(에러 처리) / finally(성공·실패 상관없이 항상 실행) 구조로 역할을 분리한다. finally는 주로 로딩 스피너를 끄거나, 아래 예시처럼 입력 모달을 닫는 뒷정리에 사용된다.

#### [ planner_fetch.js - fetchWritePlan() ]

```jsx
async function fetchWritePlan(year, month, date, title, body, file) {
    let formData = new FormData();
    formData.append("year", year);
    formData.append("month", month);
    formData.append("date", date);
    formData.append("title", title);
    formData.append("body", body);
    formData.append("file", file);

    try {
        let response = await fetch('/planner/plan', {
            method: 'POST',
            body: formData
        });

        if (!response.ok) {
            throw new Error('Network response was not ok');
        }

        let data = await response.json();

        if (!data || data.result <= 0) {
            alert("일정 등록에 문제가 발생 했습니다.");
        } else {
            alert("일정이 정상적으로 등록 되었습니다.");
            removeCalenderTr();
            addCalenderTr();
            fetchGetCurrentMonthPlans();
        }

    } catch (error) {
        alert('일정 등록에 문제가 발생 했습니다.');

    } finally {
        hideWritePlanView();   // 성공하든 실패하든 모달은 항상 닫는다
    }
}
```

## FormData 란?

- 텍스트 값과 파일(<input type="file">)을 함께 묶어서 multipart/form-data 형식으로 서버에 전송할 때 쓰는 브라우저 내장 객체다.
- fetch()의 body에 FormData 인스턴스를 그대로 넘기면, boundary가 포함된 Content-Type 헤더를 브라우저가 알아서 설정해주기 때문에 개발자가 직접 Content-Type을 지정하지 않는 것이 일반적이다.

#### [ planner_fetch.js ]

```jsx
let formData = new FormData();
formData.append("year", year);
formData.append("month", month);
formData.append("date", date);
formData.append("title", title);
formData.append("body", body);
formData.append("file", file);   // 파일도 그대로 append 가능
```

## URLSearchParams 란?

- 객체 형태의 파라미터를 year=2026&month=9 같은 GET 요청용 쿼리스트링으로 손쉽게 변환해주는 브라우저 내장 객체다.

#### [ planner_fetch.js - fetchGetCurrentMonthPlans() ]

```jsx
let reqData = {
    "year": current_year,
    "month": current_month + 1
}

let queryString = new URLSearchParams(reqData).toString();

let response = await fetch(`/planner/plans?${queryString}`, {
    method: 'GET',
    headers: {
        'Content-Type': 'application/json; charset=utf-8'
    }
});
```

function addCalenderTr() {
    console.log('addCalenderTr() CALLED!!');

    let thisCalenderStart = new Date(current_year, current_month, 1);
    let thisCalenderStartDate = thisCalenderStart.getDate();            // 현재 월의 첫 날
    let thisCalenderStartDay = thisCalenderStart.getDay();              // 현재 월의 첫 요일 2

            dateCnt++;
        }

    let tableBody = document.querySelector('#table_calender tbody');

    let dateIndex = 0;
    for (let i = 0; i < 6; i++) {

        if (i >= 5 && dates[dateIndex] === 0)
            break;

                td.appendChild(dateDiv);

                writeLink.href = "#none";

                let planList = document.createElement('ul');
                planList.className = 'plan';
                planWrap.appendChild(planList);
                td.appendChild(planWrap);

            }

            tr.appendChild(td);
            dateIndex++;

        }

        tableBody.appendChild(tr);

    }

function setPreMonth() {
    console.log('setPreMonth()');

    let yearSelect = document.querySelector('select[name="p_year"]');
    let monthSelect = document.querySelector('select[name="p_month"]');

    if (yearSelect.value == 2025 && monthSelect.value == 1) {
        alert('2025년 1월 이전은 설정할 수 없습니다.');
        return false;
    }

    let temp_year = current_year;
    let temp_month = current_month - 1;

    if (temp_month <= -1) {
        temp_year -= 1;
        temp_month = 11;
    }

    let preCalender = new Date(temp_year, temp_month, 1);

    // 현재 데이터 설정
    setCurrentCalender(
        preCalender.getFullYear(),
        preCalender.getMonth(),
        preCalender.getDate(),
        preCalender.getDay()
    )

    // UI(<select>) 렌더링
    setCurrentYearAndMonthSelectUI();

    //  UI(<tr>) 제거
    removeCalenderTr();

    // UI(<tr>) 렌더링
    addCalenderTr();

    // 해당 년/월 일정들 가져오기
    fetchGetCurrentMonthPlans();

}

function setNextMonth() {

    let temp_month = current_month + 1;

        temp_year += 1;
        temp_month = 0;

    let nextCalender = new Date(temp_year, temp_month, 1);

    // 현재 데이터 설정
    setCurrentCalender(
        nextCalender.getFullYear(),
        nextCalender.getMonth(),
        nextCalender.getDate(),
        nextCalender.getDay()
    )

    // UI(<select>) 렌더링
    setCurrentYearAndMonthSelectUI();

    //  UI(<tr>) 제거
    removeCalenderTr();

    // UI(<tr>) 렌더링
    addCalenderTr();

    // 해당 년/월 일정들 가져오기
    fetchGetCurrentMonthPlans();

}

async function fetchWritePlan(year, month, date, title, body, file) {
    console.log('fetchWritePlan() CALLED!!');

    let formData = new FormData();
    formData.append("year", year);
    formData.append("month", month);
    formData.append("date", date);
    formData.append("title", title);
    formData.append("body", body);
    formData.append("file", file);

    try {
        let response = await fetch('/planner/plan', {
            method: 'POST',
            body: formData
        });

        if (!response.ok) {
            throw new Error('Network response was not ok');
        }

        console.log('fetchWritePlan() COMMUNICATION SUCCESS!! ');

        let data = await response.json();
        console.log('data: ', data);

        if (!data || data.result <= 0) {
            alert("일정 등록에 문제가 발생 했습니다.");

        } else {
            alert("일정이 정상적으로 등록 되었습니다.");

async function fetchGetCurrentMonthPlans() {

    let reqData = {
        "year": current_year,
        "month": current_month + 1
    }

            headers: {
                'Content-Type': 'application/json; charset=utf-8'
            }
        });
