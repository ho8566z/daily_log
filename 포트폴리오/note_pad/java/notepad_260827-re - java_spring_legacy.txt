[ CalAssembler ]

package com.office.dw;

public class CalAssembler {
	
	MyCalculator myCalculator;
	CalAdd calAdd;
	CalSub calSub;
	CalMul calMul;
	CalDiv calDiv;
	
	public CalAssembler() {
		myCalculator = new MyCalculator();
		calAdd = new CalAdd();
		calSub = new CalSub();
		calMul = new CalMul();
		calDiv = new CalDiv();
		
		assemble();
	}

	private void assemble() {
		
		myCalculator.calculate(10, 5, calAdd);
		myCalculator.calculate(10, 5, calSub);
		myCalculator.calculate(10, 5, calMul);
		myCalculator.calculate(10, 5, calDiv);
	}
	
}

[ ICalculator ]

package com.office.dw;

public interface ICalculator {
	
	public int doOperation(int firstNum, int secondNum);

}

------------------------------------------------------------------------
[ CalAdd ]

package com.office.dw;

public class CalAdd implements ICalculator {

	@Override
	public int doOperation(int firstNum, int secondNum) {

		return firstNum + secondNum;
	}

}

------------------------------------------------------------------------
[ CalSub ]

package com.office.dw;

public class CalSub implements ICalculator {

	@Override
	public int doOperation(int firstNum, int secondNum) {

		return firstNum - secondNum;
	}

}

------------------------------------------------------------------------
[ CalMul ]

package com.office.dw;

public class CalMul implements ICalculator {

	@Override
	public int doOperation(int firstNum, int secondNum) {

		return firstNum * secondNum;
	}

}

------------------------------------------------------------------------
[ CalDiv ]

package com.office.dw;

public class CalDiv implements ICalculator {

	@Override
	public int doOperation(int firstNum, int secondNum) {

		return secondNum !=0 ? (firstNum / secondNum) : 0;
	}

}
